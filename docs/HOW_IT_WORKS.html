1 מה הפרויקט עושה — הרעיון הגדול

הבעיה: תלמידות עובדות בתוך Safe Exam Browser (SEB). הדפדפן חוסם:

  • Ctrl+C / Ctrl+V (העתקה והדבקה)
  • PrintScreen (צילום מסך)
  • פתיחת תוכנות מתוכו

הפתרון: תוכנת Python שרצה ב-Background — לא בתוך הדפדפן, אלא כתהליך Windows נפרד שה-SEB לא רואה. היא:

  • קוראת טקסט ישירות מהחלון הפעיל דרך Windows UIAutomation API
  • מזהה תגיות שהתלמידה כותבת בעורך הקוד
  • שולחת שאלות לאחד מכמה ספקי AI (Gemini / Groq / Mistral) ומציגה תשובה בחלון צף
  • מצלמת מסך ומעתיקה ל-Clipboard — הכל דרך Windows, לא דרך הדפדפן

    "api_key": "",
    "provider": "",
    "poll_interval": 1.0,
    "response_mode": "auto",

    "type_speed": 0.015,
    "language": "he",

VALID_PROVIDERS = ("", "gemini", "groq", "mistral")

class ConfigError(Exception):
    """נזרק אם settings.json חסר שדה או ערך לא תקין."""
    pass

def _validate(settings: dict) -> dict:
    if settings["response_mode"] not in VALID_RESPONSE_MODES:
        raise ConfigError(f"response_mode must be one of {VALID_RESPONSE_MODES}...")
    if settings["provider"] not in VALID_PROVIDERS:
        raise ConfigError(f"provider must be one of {VALID_PROVIDERS}...")
    if settings["poll_interval"] <= 0:
        raise ConfigError("poll_interval must be greater than 0")
    if settings["type_speed"] < 0:
        raise ConfigError("type_speed must be non-negative")
    return settings

def is_api_key_configured(settings: dict) -> bool:
    key = settings.get("api_key", "")
    return bool(key) and "YOUR_" not in key

def load() -> dict:
    if not _SETTINGS_FILE.exists():
        # אין settings.json? מעתיקים מ-settings.example.json (או יוצרים defaults)
        if _EXAMPLE_FILE.exists():
            shutil.copy(_EXAMPLE_FILE, _SETTINGS_FILE)
        else:
            _SETTINGS_FILE.write_text(json.dumps(_DEFAULTS, ...))

    with open(_SETTINGS_FILE, encoding="utf-8") as f:
        raw = json.load(f)
    merged = {**_DEFAULTS, **raw}
    return _validate(merged)   # זורק ConfigError אם משהו לא תקין

Settings = load()   # רץ מיד בטעינה. כל קובץ אחר: from config import Settings

6 reader.py — קריאת טקסט מ-SEB

זה הפתרון הטכני המרכזי, ולא השתנה בין הגרסאות. Windows UIAutomation הוא API נגישות (accessibility) של מערכת ההפעלה — נועד לקוראי מסך. SEB לא חוסם אותו כי הוא חלק מ-Windows עצמו, לא מהדפדפן.

reader.py — הפונקציה הראשית
def read_focused_text() -> Optional[str]:
    try:
        import uiautomation as auto
    except ImportError:
        print("[reader] uiautomation not installed...")
        return None

    # ── שלב 0: בדוק שהחלון הפעיל לא שלנו ──────────────
    fg = auto.GetForegroundControl()
    if fg and _should_ignore(fg.Name):
        return None    # Terminal / Notepad שלנו — תתעלם

    # ── שלב 1: קריאה מהירה מהאלמנט בפוקוס ─────────────
    ctrl = auto.GetFocusedControl()
    if ctrl:
        v = _extract(ctrl)
        if v and "@@" in v:
            return v    # מצאנו! הכי מהיר

        # ── שלב 2: עלייה בהורים (Monaco עוטף טקסט ב-divs) ──
        parent = ctrl.GetParentControl()
        for _ in range(MAX_PARENT_WALK_UP):    # מ-constants.py, ברירת מחדל 5
            if parent is None: break
            v = _extract(parent)
            if v and "@@" in v: return v
            parent = parent.GetParentControl()

    # ── שלב 3: חיפוש בעץ המלא ──────────────────────────
    if fg:
        return _search_tree(fg, depth=0)   # עד MAX_TREE_SEARCH_DEPTH (7)
למה להתעלם מ-Notepad? אם TAGS_GUIDE.txt פתוח ב-Notepad, הוא מלא @@תגיות@@ והסוכן יתבלבל. לכן כל קובץ .txt וכל Notepad מתעלמים (_IGNORE_TITLES).

למה לעלות בהורים? Monaco editor (של Ultracode) בנוי מ-divs בתוך divs. הפוקוס עלול להיות על div פנימי ריק. הפתרון: עוברים עד MAX_PARENT_WALK_UP הורים למעלה עד שמוצאים את הטקסט.

מה השתנה? אותה לוגיקה בדיוק — רק ה-5 וה-7 הקשיחים הפכו לקבועים בשם ב-constants.py, ונוספה בדיקת ImportError מפורשת עם הודעה ברורה אם uiautomation לא מותקן.

7 detector.py — זיהוי תגיות

detector.py — regex + peek_tag()
# מוצא @@TAGNAME: תוכן@@ (הנקודותיים אופציונלי, re.DOTALL = כולל שורות חדשות)
# רגקס אחד בלבד מטפל גם בתגיות עם תוכן וגם בתגיות "ריקות" כמו @@SCREENSHOT@@
_TAG_RE = re.compile(r'@@([A-Za-z]+)\s*:?\s*(.*?)@@', re.DOTALL)

KNOWN_TAGS = {"ASK", "SOLVE", "FIX", "COPY", "COPYALL",
              "COPYABOVE", "SCREENSHOT", "SOLVEALL", "ASKALL"}

def peek_tag(text: str) -> tuple[str, str] | None:
    """מוצא את התגית האחרונה בטקסט. לא מעדכן state."""
    candidates = []
    for m in _TAG_RE.finditer(text):
        tag = m.group(1).upper().strip()
        content = m.group(2).strip()
        if tag in KNOWN_TAGS:
            candidates.append((m.start(), tag, content))
    # max לפי start = התגית הכי קרובה לסוף הטקסט (האחרונה)
    if not candidates: return None
    _, tag_type, content = max(candidates, key=lambda x: x[0])
    return tag_type, content
detector.py — מניעת כפילויות עם MD5 hash
_processed_hash: str | None = None   # זוכר תגית אחרונה שעובדה

def _compute_hash(tag_type: str, content: str) -> str:
    return hashlib.md5(f"{tag_type}:{content}".encode()).hexdigest()

def is_new(tag_type, content) -> bool:
    return _compute_hash(tag_type, content) != _processed_hash

def mark_processed(tag_type, content):
    global _processed_hash
    _processed_hash = _compute_hash(tag_type, content)
    # נקרא רק אחרי שהתגית נשלחה בפועל — לא בזמן הזיהוי
מה השתנה? הגרסה הקודמת הכילה שני regex-ים: _TAG_RE הכללי, ועוד _TAG_BARE נפרד לתגיות ריקות (SCREENSHOT/COPYALL/COPYABOVE/SOLVEALL). בבדיקה התברר ש-_TAG_RE לבד כבר תופס גם את התגיות הריקות (content ריק) — כך ש-_TAG_BARE היה תמיד מייצר candidate כפול וזהה, ו-max() פשוט התעלם ממנו. הוסר קוד מת, אין שינוי בהתנהגות.

מנגנון ה-Debounce

הבעיה: תלמידה מקלידה @@ASK: למ — עוד לא סיימה. אם נשלח ל-AI על כל הקשה נבזבז טוקנים ונקבל תשובות שגויות. הפתרון: מחכים שהתגית תהיה יציבה 1.5 שניות (ללא שינוי מהגרסה הקודמת).

main.py — לולאת ה-debounce (מקוצר)
while True:
    text = read_focused_text()
    if text and "@@" in text:
        result = peek_tag(text)
        if result:
            tag_type, content = result
            key = f"{tag_type}:{content}"

            if not is_new(tag_type, content) and tag_type not in REPEATABLE_TAGS:
                pass   # כבר עובד — דלג
            elif key != pending_key:
                pending_key, pending_since, pending_data = key, time.time(), (tag_type, content, text)
            else:
                debounce = DEBOUNCE_BARE if not content else DEBOUNCE_WITH_CONTENT
                if time.time() - pending_since >= debounce:
                    mark_processed(tag_type, content)
                    threading.Thread(target=_safe_dispatch, args=pending_data, daemon=True).start()

    time.sleep(MAIN_LOOP_POLL_INTERVAL)   # 0.3 שניות

_safe_dispatch() — עוטף את tag_handlers.dispatch()

def _safe_dispatch(tag_type: str, content: str, full_text: str):
    """אם handler קורס — לא נעלם בשקט, אלא מודפס ומוצג בחלון ERROR"""
    try:
        dispatch(tag_type, content, full_text)   # מ-tag_handlers.py
    except Exception as e:
        print(f"[Agent] Unhandled error: {e}")
        ui_window.show("ERROR", f"שגיאה לא צפויה:\n{e}")
מה השתנה הכי הרבה? ב-main.py הישן, פונקציית _handle() אחת הכילה if/elif ל-9 סוגי תגיות — כ-200 שורות. עכשיו main.py רק מריץ את הלולאה וקורא ל-dispatch() מ-tag_handlers.py (סעיף 10), שמחזיק handler נפרד לכל תגית. הלוגיקה עצמה — מה כל תגית עושה — לא השתנתה כלל.

9 ai_provider.py — שכבת ה-AI (Gemini / Groq / Mistral)

תומך בשלושה ספקי AI, נבחר אוטומטית לפי צורת ה-API key. כל בקשה עצמאית (ללא היסטוריית שיחה) לחיסכון בטוקנים.

ai_provider.py — בחירת ספק
def detect_provider() -> str:
    explicit = Settings.get("provider", "").lower()
    if explicit in ("groq", "gemini", "mistral"):
        return explicit
    key = Settings.get("api_key", "")
    if key.startswith("gsk_"): return "groq"
    if key.startswith(("AIzaSy", "AQ.")): return "gemini"
    return "mistral"

שלושת הספקים יורשים מ-AIProvider, שמחזיקה את לולאת ה-retry פעם אחת בלבד (ולא משוכפלת בכל ספק כמו בגרסה הראשונה של הרפקטורינג):

ai_provider.py — retry אחיד לכל הספקים
class RateLimitError(Exception):
    """ספק זורק את זה כדי לאותת ל-generate() לנסות שוב"""
    pass

class AIProvider(ABC):
    @abstractmethod
    def _request(self, tag_type: str, user_msg: str):
        """ניסיון אחד. זורק RateLimitError כדי להפעיל retry."""

    def generate(self, tag_type: str, user_msg: str) -> str:
        for attempt in range(1, RETRY_ATTEMPTS + 1):
            try:
                return self._request(tag_type, user_msg)
            except RateLimitError:
                if attempt < RETRY_ATTEMPTS:
                    print(f"[AI] Rate limit — ממתין {RETRY_WAIT_SECONDS}s... ({attempt}/{RETRY_ATTEMPTS})")
                    time.sleep(RETRY_WAIT_SECONDS)
                else:
                    raise

כל ספק מממש רק את _request() — הקריאה הספציפית ל-API שלו — ומתרגם 429/rate-limit ל-RateLimitError כדי שה-retry המשותף יתפוס אותו:

ai_provider.py — דוגמה: MistralProvider (הכי קצר)
class MistralProvider(AIProvider):
    def _request(self, tag_type: str, user_msg: str) -> str:
        import requests
        system, max_tokens = _system_and_limit(tag_type)
        resp = requests.post(
            "https://api.mistral.ai/v1/chat/completions",
            headers={"Authorization": f"Bearer {Settings['api_key']}"},
            json={"model": Settings.get("model", "mistral-small-latest"),
                  "messages": [{"role": "system", "content": system},
                               {"role": "user", "content": user_msg}],
                  "max_tokens": max_tokens, "temperature": 0.3},
            timeout=30,
        )
        if resp.status_code == 429:
            raise RateLimitError(f"429: {resp.text}")
        resp.raise_for_status()
        return resp.json()["choices"][0]["message"]["content"].strip()

    if tag_type == "FIX" and context:
        user_msg = f"Code:\n{context}\n\nFix instruction: {content}"
    else:
        user_msg = content
    if not user_msg.strip(): raise ValueError("Request cannot be empty")
    return get_provider().generate(tag_type, user_msg)

def type_text(text: str):
    delay = Settings.get("type_speed", 0.015)

    # ניסיון 1: keyboard library — תומך Unicode ועברית
    try:
        import keyboard
        keyboard.write(text, delay=delay)
        return
    except ImportError:
        pass

    # Fallback: pyautogui — עלול לאבד יוניקוד
    try:
        import pyautogui
        pyautogui.write(text, interval=delay)
    except ImportError:
        print("[keyboard_input] אין keyboard ואין pyautogui")

def copy_to_clipboard(text: str) -> bool:
    """pyperclip קודם, Tk זמני כ-fallback. מחזיר True/False בהצלחה."""
    try:
        import pyperclip
        pyperclip.copy(text)
        return True
    except Exception as e:
        print(f"[clipboard] pyperclip failed ({e}), trying Tk fallback...")

    try:
        import tkinter as tk
        r = tk.Tk(); r.withdraw()
        r.clipboard_clear(); r.clipboard_append(text); r.update()
        r.destroy()
        return True
    except Exception as e:
        print(f"[clipboard] Tk fallback also failed: {e}")
        return False
למה זה טוב יותר? עכשיו כל קריאה ל-COPY/COPYALL/COPYABOVE יודעת אם ההעתקה באמת הצליחה ומציגה ERROR אם לא — בעבר תמיד הונח שההעתקה הצליחה.
